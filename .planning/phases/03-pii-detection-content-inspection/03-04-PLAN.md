# Success criteria from ROADMAP.md
1. An AI prompt containing a name, email, phone number, SSN, or address is intercepted and the PII is replaced with category-tagged placeholders (e.g., `[PII:NAME]`, `[PII:EMAIL]`) before reaching the vendor
2. An AI prompt containing credit card numbers, bank accounts, or API keys is intercepted and the sensitive data is replaced with appropriate category tags (e.g., `[FINANCIAL:CARD]`, `[SECRET:API_KEY]`)
3. A streaming AI response containing PII is inspected via the sliding window token buffer (5-10 tokens held back) and sensitive content is redacted before reaching the client
4. The kernel can sever a streaming connection mid-response and inject `[REDACTED BY INTERDICT POLICY: {RULE_NAME}]` when a severe policy violation is detected in the response stream
5. Custom enterprise patterns (client names, matter numbers, case codes) loaded from policy configuration are detected and redacted alongside built-in patterns

```rust
// Success Criterion 1: PII detection and placeholder replacement
#[test]
fn test_pii_detection_and_placeholder_replacement() {
    let registry = Arc::new(PatternRegistry {
        patterns: default_patterns(),
        version: 1,
    });
    let redactor = Arc::new(RedactionEngine::new());
    let config = Arc::new(PolicyConfig::default());
    
    let inspector = ContentInspector::new(registry, redactor, config);
    
    // Test email detection
    let result = inspector.inspect_request(b"Contact user@example.com for details");
    assert!(matches!(result.action, VerdictAction::Redact));
    assert!(result.detections.contains(&"EMAIL".to_string()));
    let redacted = String::from_utf8_lossy(result.redacted_content.unwrap().as_ref());
    assert!(redacted.contains("[") && redacted.contains("EMAIL") && redacted.contains("]"));
    
    // Test phone number detection
    let result = inspector.inspect_request(b"Call me at 555-123-4567");
    assert!(matches!(result.action, VerdictAction::Redact));
    assert!(result.detections.contains(&"PHONE".to_string()));
    
    // Test SSN detection
    let result = inspector.inspect_request(b"SSN: 123-45-6789");
    assert!(matches!(result.action, VerdictAction::Redact));
    assert!(result.detections.contains(&"SSN".to_string()));
    
    // Test address detection
    let result = inspector.inspect_request(b"Address: 123 Main St, Springfield, IL 62701");
    assert!(matches!(result.action, VerdictAction::Redact));
    assert!(result.detections.contains(&"ADDRESS".to_string()));
}

async fn test_streaming_inspection_with_buffer() {
    let registry = Arc::new(PatternRegistry {
        patterns: default_patterns(),
        version: 1,
    });
    let redactor = Arc::new(RedactionEngine::new());
    let detector = Arc::new(StreamingDetector::new(registry, redactor));
    
    let relay = InspectingRelay::new(detector, BufferPreset::Medium);
    
    let (input_tx, input_rx) = mpsc::channel(32);
    let (output_tx, mut output_rx) = mpsc::channel(32);
    
    // Simulate streaming response with PII spread across chunks
    tokio::spawn(async move {
        input_tx.send(Bytes::from("The email ")).await.unwrap();
        input_tx.send(Bytes::from("address is ")).await.unwrap();
        input_tx.send(Bytes::from("user@")).await.unwrap();  // Partial email
        input_tx.send(Bytes::from("example.com ")).await.unwrap();  // Complete email
        input_tx.send(Bytes::from("for contact")).await.unwrap();
        drop(input_tx);
    });
    
    tokio::spawn(async move {
        relay.relay_with_inspection(input_rx, output_tx, vec![], String::new())
            .await
            .unwrap();
    });
    
    // Collect output
    let mut collected = Vec::new();
    while let Some(chunk) = output_rx.recv().await {
        collected.push(chunk);
    }
    
    let result: String = collected.iter()
        .flat_map(|b| String::from_utf8_lossy(b).chars().collect::<Vec<_>>())
        .collect();
    
    // Email should be redacted
    assert!(result.contains("[") && (result.contains("EMAIL") || result.contains("REDACTED")));
    
    // Buffer should have held back tokens (5-10) during processing
    // This is verified by the test not failing and redaction being applied
}

#[tokio::test]
async fn test_stream_severing_on_severe_violation() {
    let registry = Arc::new(PatternRegistry {
        patterns: default_patterns(),
        version: 1,
    });
    let redactor = Arc::new(RedactionEngine::new());
    let detector = Arc::new(StreamingDetector::new(registry, redactor));
    
    let relay = InspectingRelay::new(detector, BufferPreset::Small);
    
    let (input_tx, input_rx) = mpsc::channel(32);
    let (output_tx, mut output_rx) = mpsc::channel(32);
    
    // Simulate streaming response with AWS key
    tokio::spawn(async move {
        input_tx.send(Bytes::from("Here is your key: ")).await.unwrap();
        input_tx.send(Bytes::from("AKIAIOSFODNN7EXAMPLE")).await.unwrap();
        drop(input_tx);
    });
    
    let sever_categories = vec!["AWS_KEY".to_string()];
    let sever_message = "[REDACTED BY INTERDICT POLICY: SECRET_DETECTION]".to_string();
    
    let result = relay.relay_with_inspection(
        input_rx,
        output_tx,
        sever_categories,
        sever_message.clone(),
    ).await;
    
    // Should have severed
    assert!(result.is_err());
    assert!(result.unwrap_err().contains("stream severed"));
    
    // Should have injected policy message
    let injected = output_rx.recv().await.unwrap();
    let text = String::from_utf8_lossy(&injected);
    assert!(text.contains("REDACTED BY INTERDICT POLICY"));
    assert!(text.contains("SECRET_DETECTION"));
}

// Success Criterion 5: Custom pattern detection
#[test]
fn test_custom_pattern_detection() {
    // Test regex-based custom pattern
    let custom_regex = CustomPattern {
        name: "matter_number".to_string(),
        category: "MATTER".to_string(),
        source: PatternSource::Regex(r"\bM\d{6}\b".to_string()),
        threshold: 0.9,
    };
    
    let rule = custom_regex.compile().unwrap();
    assert!(rule.pattern.is_match("Matter M123456 is confidential"));
    assert!(!rule.pattern.is_match("Matter M12345 is confidential"));
    
    // Test example-based custom pattern
    let custom_examples = CustomPattern {
        name: "client_names".to_string(),
        category: "CLIENT".to_string(),
        source: PatternSource::Examples(vec![
            "Acme Corp".to_string(),
            "Globex Ltd".to_string(),
            "Initech Inc".to_string(),
        ]),
        threshold: 0.95,
    };
    
    let rule = custom_examples.compile().unwrap();
    assert!(rule.pattern.is_match("Acme Corp"));
    assert!(rule.pattern.is_match("Globex Ltd"));
    assert!(rule.pattern.is_match("Initech Inc"));
    assert!(!rule.pattern.is_match("Unknown Corp"));
    
    // Test integration with inspector
    let mut patterns = default_patterns();
    patterns.push(custom_regex.compile().unwrap());
    patterns.push(custom_examples.compile().unwrap());
    
    let registry = Arc::new(PatternRegistry {
        patterns,
        version: 1,
    });
    let redactor = Arc::new(RedactionEngine::new());
    let config = Arc::new(PolicyConfig::default());
    
    let inspector = ContentInspector::new(registry, redactor, config);
    
    let result = inspector.inspect_request(b"Matter M789012 for Acme Corp");
    assert!(matches!(result.action, VerdictAction::Redact));
    assert!(result.detections.contains(&"MATTER".to_string()));
    assert!(result.detections.contains(&"CLIENT".to_string()));
}
```

```rust
use criterion::{black_box, criterion_group, criterion_main, Criterion, BenchmarkId};
use interdict_kernel::policy::patterns::{PatternRegistry, default_patterns};
use interdict_kernel::policy::streaming::StreamingDetector;
use interdict_kernel::policy::redaction::RedactionEngine;
use std::sync::Arc;

fn benchmark_pattern_detection(c: &mut Criterion) {
    let registry = Arc::new(PatternRegistry {
        patterns: default_patterns(),
        version: 1,
    });
    let redactor = Arc::new(RedactionEngine::new());
    let detector = StreamingDetector::new(registry, redactor);
    
    let samples = vec![
        ("clean_text", "This is a completely clean message with no sensitive data"),
        ("with_email", "Contact me at john.doe@example.com for more information"),
        ("with_phone", "You can reach me at 555-123-4567 during business hours"),
        ("with_credit_card", "Payment via card 4532-1488-0343-6467 was processed"),
        ("mixed_pii", "John Doe, email john@example.com, phone 555-1234, SSN 123-45-6789"),
    ];
    
    let mut group = c.benchmark_group("pattern_detection");
    
    for (name, content) in samples {
        group.bench_with_input(BenchmarkId::from_parameter(name), content, |b, content| {
            b.iter(|| {
                detector.scan(black_box(content))
            });
        });
    }
    
    group.finish();
}

fn benchmark_redaction_application(c: &mut Criterion) {
    let registry = Arc::new(PatternRegistry {
        patterns: default_patterns(),
        version: 1,
    });
    let redactor = Arc::new(RedactionEngine::new());
    let detector = StreamingDetector::new(registry, redactor);
    
    let content = "Contact john.doe@example.com or call 555-123-4567 for assistance";
    
    c.bench_function("redaction_apply", |b| {
        // Pre-detect
        let scan_result = detector.scan(content);
        let detections = match scan_result {
            interdict_kernel::policy::streaming::ScanResult::FullMatch(d) => d,
            _ => vec![],
        };
        
        b.iter(|| {
            detector.apply_redaction(black_box(content), black_box(detections.clone()))
        });
    });
}

fn benchmark_buffer_operations(c: &mut Criterion) {
    use interdict_kernel::policy::streaming::{AdaptiveTokenBuffer, BufferPreset};
    use bytes::Bytes;
    
    c.bench_function("buffer_push_emit", |b| {
        b.iter(|| {
            let mut buffer = AdaptiveTokenBuffer::new(BufferPreset::Medium);
            for i in 0..20 {
                buffer.push(Bytes::from(format!("token{} ", i)));
                buffer.emit_oldest();
            }
        });
    });
}

criterion_group!(
    benches,
    benchmark_pattern_detection,
    benchmark_redaction_application,
    benchmark_buffer_operations
);
criterion_main!(benches);
```
