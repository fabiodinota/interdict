pub fn inspect_request(&self, content: &[u8]) -> InspectionResult {
    let text = String::from_utf8_lossy(content);

    // Compute SHA-256 before any modification
    let mut hasher = Sha256::new();
    hasher.update(content);
    let original_hash = format!("{:x}", hasher.finalize());

    // PLCY-11: Check for prompt injection / jailbreak FIRST
    // Injection attempts are treated as block-level violations
    let injections = self.injection_detector.detect(&text);
    if !injections.is_empty() {
        let categories: Vec<String> = injections
            .iter()
            .map(|d| format!("INJECTION:{:?}", d.kind))
            .collect();
        return InspectionResult {
            action: VerdictAction::Block,
            redacted_content: None,
            detections: categories,
            original_hash,
            reason: "prompt injection or jailbreak attempt detected".to_string(),
        };
    }

    // ... rest of existing pattern scanning logic unchanged ...
}
