```json
{
  "slug": "string",           // kebab-case directory name
  "name": "string",           // Human-readable framework name
  "description": "string",    // Full framework description
  "jurisdiction": "string",   // Geographic scope (EU, US, SG, IN, CN, CA, GCC)
  "version": "string",        // Year or version number
  "policies": [
    {
      "file": "policies/{name}.rego",          // Relative path to .rego file
      "requirement_ref": "string",              // Article/section reference
      "requirement_description": "string",      // Plain English description
      "is_required": true,                      // Always true for seed policies
      "sort_order": 1                           // 1-indexed sequential
    }
  ]
}
```

Rego policy template (from existing policies):
```rego
# {Framework} - {Article/Section}: {Short title}
#
# {Description of what this policy enforces}

package interdict.policy.verdict

import rego.v1

default verdict := {"action": "allow", "reason": "", "redactions": []}

# Block when violation detected
verdict := result if {
    input.direction == "request"
    content := lower(input.content)
    violation_pattern(content)
    not exemption_marker(content)
    result := {
        "action": "block",
        "reason": "{Framework} {Article} - {Category}: {Specific violation description}",
        "redactions": [],
    }
}

# Violation patterns (5-8 helper rules using contains())
violation_pattern(content) if { contains(content, "...") }

# Exemption markers (3-5 helper rules using contains())
exemption_marker(content) if { contains(content, "...") }
```
